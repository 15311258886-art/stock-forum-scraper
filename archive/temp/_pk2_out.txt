function learnSubItems(fallback) {
  const tracks = toArr(KB.tracks && KB.tracks.docs);
  if (!tracks.length) return fallback;
  const items = tracks.map((d) => ({
    tab: "learn_" + d.doc_key,
    label: d.track || d.title || d.doc_key,
    docKey: d.doc_key,
  }));
  items.push({ tab: "learn_tracks", label: "学习轨道管理" });
  return items;
}

function renderCohold(d) {
  const box = $("#p-cohold");
  if (!box) return;
  if (!d || typeof d !== "object" || Array.isArray(d)) {
    box.innerHTML = `<div class="empty">近5日共同买入：接口返回类型不符，已降级为空。</div>`;
    return;
  }
  COHOLD.data = d;
  box.innerHTML = `
    <div class="hint" style="margin-bottom:10px">
      按「窗口内 ≥<b>${esc(d.min_holders ?? COHOLD.min)}</b> 位选手<b>同向操作</b>同一标的」聚合，<b>共买</b>（红）与<b>共卖</b>（绿）分两组列出。
      与左侧「共同持仓共识」<b>口径不同</b>：共识看的是<b>现在还持有</b>（持仓快照），本板块看的是<b>最近 ${esc(d.days)} 天谁同时动手</b>（动作发生时点）——
      所以本板块能露出「窗口内已卖出」这类共识板块看不到的行。数据合并 <b>positions</b>（仓位占比/买价/现价）与 <b>trades</b>（<b>成交数量</b>、更精确的成交时刻）两张表。
    </div>
    ${coholdWindowBar(d)}
    <div class="coh-legend">
      <span class="badge bull">共同买入</span> <span class="coh-lbl">红 = 买（A股惯例）</span>
      <span class="coh-sep"></span>
      <span class="badge bear">共同卖出</span> <span class="coh-lbl">绿 = 卖</span>
    </div>
    ${coholdTable(d, "buy")}
    ${coholdTable(d, "sell")}
    <div class="hint" style="margin-top:12px">
      <b>数量</b>来自 trades 表的成交记录——positions 表没有这一列，所以只有持仓快照、没有成交明细的行会显示「—」，<b>不会用 0 或估算值顶替</b>。
      <b>浮盈</b>用同一笔的买价与现价计算，不跨笔混算。<b>持有状态</b>取「买点之后是否存在该选手对该标的的卖出记录」，有则记为「买入后已卖出」。
      ${freshLegend()}
    </div>`;
  bindCohold();
  applyFreshness();
}

def api_cohold_recent(days: int = 5, min_holders: int = 2,
                      anchor: str | None = None) -> dict:
    """近 N 日「多人同向操作」：共买 / 共卖两组，附买入时间、价格、数量、浮盈与持有状态。

    与 /api/consensus 的区别：那边看的是**当前持仓快照**（现在还有多少人拿着），
    这边看的是**最近 N 天谁同时动手了**（动作发生时点）。两者互补，不可互替。

    名称补全在服务端统一做（与其它接口一致），避免前端各自回退出不一致的显示名。
    """
    d = DB.cohold_recent(days=days, min_holders=min_holders, anchor=anchor)
    for grp in ("buy", "sell"):
        for x in d.get(grp) or []:
            n = _named(x["symbol"], x.get("symbol_name") or "")
            x["symbol_name"] = n["name"]
            x["company"] = n["company"]
            x["market"] = n["market"]
            x["name_missing"] = n["name_missing"]
            if n["name"]:
                x["label"] = f"{n['name']}（{x['symbol']}）"
            for p in x.get("players") or []:
                p["symbol_name"] = n["name"] or x["symbol_name"] or ""
    return d
