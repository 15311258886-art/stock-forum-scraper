function renderPersons(list) {
  const all = toArr(list);
  if (!all.length) { $("#p-persons").innerHTML = `<div class="empty">暂无人物数据</div>`; return; }
  // P2-5：接口默认只回 100 条，此处再取前 80 条。文案必须如实说明截断，不能写「不遗漏任何一个名称」。
  const total = all.length, shown = Math.min(total, 80);
  const rows = all.slice(0, shown).map((p) => {
    const plats = Array.isArray(p.sites) && p.sites.length ? p.sites : (p.platform ? [p.platform] : []);
    return `<tr data-name="${escAttr(p.name)}" style="cursor:pointer"><td>${esc(p.name)}</td><td>${sourceChannel(plats, p.tier)}</td><td><span class="chip-s">${esc(p.role === "mention" ? "@提及" : "作者")}</span></td><td class="num">${p.mention_count ?? "—"}</td>${refCell(plats)}</tr>`;
  }).join("");
  $("#p-persons").innerHTML = `<div class="hint" style="margin-bottom:10px">每个名称均标注其<b>来源渠道</b>：<b>平台</b>（淘股吧/东方财富/雪球/同花顺，即该名称来自的论坛） + <b>所属层级</b>（A~F，入选选手才有）。按出现次数降序排列，<b>此处列出前 ${shown} 位</b>（本次接口共返回 ${total} 位）；完整名单请以数据库导出为准。${freshLegend()}</div>
    <table><thead><tr><th>姓名/昵称</th><th>来源渠道（平台 · 层级）</th><th>角色</th><th class="num">出现次数</th><th>最后更新</th></tr></thead><tbody>${rows}</tbody></table>`;
  bindRows();
  applyFreshness();
}

function renderCohold(d) {
  const box = $("#p-cohold");
  if (!box) return;
  if (!d || typeof d !== "object" || Array.isArray(d)) {
    box.innerHTML = `<div class="empty">近5日共同买入：接口返回类型不符，已降级为空。</div>`;
    return;
  }
  COHOLD.data = d;
  box.innerHTML = `
    <div class="hint" style="margin-bottom:10px">
      按「窗口内 ≥<b>${esc(d.min_holders ?? COHOLD.min)}</b> 位选手<b>同向操作</b>同一标的」聚合，<b>共买</b>（红）与<b>共卖</b>（绿）分两组列出。
      与左侧「共同持仓共识」<b>口径不同</b>：共识看的是<b>现在还持有</b>（持仓快照），本板块看的是<b>最近 ${esc(d.days)} 天谁同时动手</b>（动作发生时点）——
      所以本板块能露出「窗口内已卖出」这类共识板块看不到的行。数据合并 <b>positions</b>（仓位占比/买价/现价）与 <b>trades</b>（<b>成交数量</b>、更精确的成交时刻）两张表。
    </div>
    ${coholdWindowBar(d)}
    <div class="coh-legend">
      <span class="badge bull">共同买入</span> <span class="coh-lbl">红 = 买（A股惯例）</span>
      <span class="coh-sep"></span>
      <span class="badge bear">共同卖出</span> <span class="coh-lbl">绿 = 卖</span>
    </div>
    ${coholdTable(d, "buy")}
    ${coholdTable(d, "sell")}
    <div class="hint" style="margin-top:12px">
      <b>数量</b>来自 trades 表的成交记录——positions 表没有这一列，所以只有持仓快照、没有成交明细的行会显示「—」，<b>不会用 0 或估算值顶替</b>。
      <b>浮盈</b>用同一笔的买价与现价计算，不跨笔混算。<b>持有状态</b>取「买点之后是否存在该选手对该标的的卖出记录」，有则记为「买入后已卖出」。
      ${freshLegend()}
    </div>`;
  bindCohold();
  applyFreshness();
}
