=== 各 tab 的渲染函数与接口调用 ===
  tab=dashboard      ->  if (DASH.data) renderDashboard(); else loadDashboard().catch((e) => reportLoadError("今日看板", e)); 
  tab=lhb            ->  if (LHB.data) renderLhb(); else loadLhb().catch((e) => reportLoadError("游资龙虎榜", e)); 
  tab=lhb_sectors    ->  if (LBHS.data) renderLhbSectors(); else loadLhbSectors().catch((e) => reportLoadError("板块动向", e)); 
  tab=niusan         ->  if (NIUSAN.data) renderNiusan(); else loadNiusan().catch((e) => reportLoadError("牛散追踪", e)); 
  tab=star_moves     ->  if (STARM.data) renderStarMoves(); else loadStarMoves().catch((e) => reportLoadError("牛散动向", e)); 

=== 每个 tab 用到的 /api/ 接口 ===
  loadStockQuote         -> /api/quote?symbol=${encodeURIComponent
  loadKline              -> /api/kline?symbol=${encodeURIComponent
  loadLhb                -> /api/lhb
  loadNiusan             -> /api/niusan
  loadLhbSectors         -> /api/dashboard, /api/lhb_sectors
  loadDashboard          -> /api/dashboard
  loadStarMoves          -> /api/star_moves
  loadRawHtml            -> /api/detail?kind=${kind}&id=${id}&full=1
  loadAll                -> /api/dashboard, /api/summary
  loadSchedulerStatus    -> /api/scheduler/status
  loadRefreshStatus      -> /api/refresh_status
  loadKbTracks           -> /api/kb/list
