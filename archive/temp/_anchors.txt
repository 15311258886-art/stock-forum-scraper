function ensurePanel(tab) {
  let p = $("#p-" + tab);
  if (!p) {
    p = document.createElement("div");
    p.className = "panel";
    p.id = "p-" + tab;
    const host = document.querySelector(".content");
    if (host) host.appendChild(p); else return null;
  }
  return p;
}
function ensureKbDoc(force) {
  const box = ensurePanel("learn_zdx_stock");
  if (!box) return;
  if (KB.doc && !force) { renderKbDoc(); return; }
  if (KB.loading) return;
  KB.loading = true;
  if (!KB.doc) box.innerHTML = `<div class="empty">正在读取知识库…</div>`;
  api("/api/kb/doc?key=" + encodeURIComponent(KB.key))
    .then((d) => { KB.doc = d; KB.error = ""; renderKbDoc(); })
    .catch((e) => { KB.error = apiErrText(e); box.innerHTML = `<div class="empty">知识库读取失败：${esc(KB.error)}</div>`; })
    .finally(() => { KB.loading = false; });
}
